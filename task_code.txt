-------- code: make a new file embed.py with this code --------

import numpy as np
from fastembed import TextEmbedding

MODEL = "sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"
model = TextEmbedding(MODEL, cache_dir="models")

sentences = ["I want a refund.",
             "How do I get my money back?",
             "أريد استرداد أموالي",                  # Arabic: I want my money back
             "The library opens at 8:00."]
vectors = np.array(list(model.embed(sentences)))
vectors = vectors / np.linalg.norm(vectors, axis=1, keepdims=True)
print(vectors.shape)                                  # (sentences, numbers)
print(np.round(vectors @ vectors.T, 2))               # similarity of every pair

-------- code: make a new file search.py with this code --------

import numpy as np
from lab import SECTIONS, embed, evaluate

section_vectors = embed(SECTIONS)                    # 70 x 384 numbers

def meaning_search(question):
    scores = section_vectors @ embed([question])[0]  # one score per section
    best = np.argsort(-scores)[:10]                  # 10 best, best first
    return [SECTIONS[i] for i in best]

evaluate(meaning_search)

-------- code: add this code at the end of search.py --------

import re
from rank_bm25 import BM25Okapi

def words(text):                                     # "PR-12?" -> ["pr", "12"]
    return re.findall(r"\w+", text.lower())

bm25 = BM25Okapi([words(s) for s in SECTIONS])       # word counts per section

def word_search(question):
    scores = bm25.get_scores(words(question))
    best = np.argsort(-scores)[:10]
    return [SECTIONS[i] for i in best if scores[i] > 0]   # a word must match

def hybrid_search(question):
    points = {}                                      # section -> points
    for search in (meaning_search, word_search):
        for rank, section in enumerate(search(question), 1):
            points[section] = points.get(section, 0) + 1 / (60 + rank)
    return sorted(points, key=points.get, reverse=True)[:10]

evaluate(word_search)
evaluate(hybrid_search)

-------- code: add this code at the end of search.py --------

from fastembed.rerank.cross_encoder import TextCrossEncoder
from lab import QUESTIONS

reranker = TextCrossEncoder("BAAI/bge-reranker-v2-m3-int8", cache_dir="models")

def rerank_search(question):
    candidates = hybrid_search(question)                   # the top 10
    scores = list(reranker.rerank(question, candidates))   # read each pair
    best = np.argsort(scores)[::-1]                        # highest first
    return [candidates[i] for i in best]

evaluate(rerank_search)

question = QUESTIONS[18]["question"]                       # q19, in Arabic

    prompt += f"[{i}] {section}\n\n"
prompt += "Question: " + question

-------- code: make a new file qdrant_search.py with this code --------

from qdrant_client import QdrantClient, models

from lab import SECTIONS, evaluate

client = QdrantClient(path="qdrant_db")             # stored in a folder, no server
MODEL = "sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"

def doc(text):                                      # Qdrant embeds it with fastembed
    return models.Document(text=text, model=MODEL, options={"cache_dir": "models"})

if not client.collection_exists("guide"):           # build it on the first run only
    client.create_collection("guide", vectors_config=models.VectorParams(
        size=client.get_embedding_size(MODEL), distance=models.Distance.COSINE))
    client.upsert("guide", [models.PointStruct(id=i, vector=doc(text), payload={"text": text})
                            for i, text in enumerate(SECTIONS)])

def qdrant_search(question):
    hits = client.query_points("guide", query=doc(question), limit=10).points
    return [hit.payload["text"] for hit in hits]

evaluate(qdrant_search)
client.close()
